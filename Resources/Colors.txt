# Colors.txt
#
# Named color palettes referenced by Effects.txt, Monsters.txt, and Items.txt.
# Each entry gives a name to one or more RGBA colors (animation frames).
#
# Color <name>
# {
#     Color   <R,G,B,A>                                  # single color
# }
#
# Color <name>
# {
#     Color   <<R,G,B,A>;<R,G,B,A>;<R,G,B,A>>            # animating sequence
# }
#
# Components are 0-255.  Alpha of 255 is fully opaque.

}
