# Effects.txt
#
# Items, monsters, and spells all reference these by name.
#
# Each Effect entry is of the following form:
#
# Effect <effect-name>
# {
#    variable-name    value
#     ...
# }
#
# Note: delimit <string> values with angle brackets <>
# Where variable-name/value are name/value pairs
# from the following list:
#
# Type        string    EFFECT_TYPE_* verb (HIT, HEAL, CREATE, DESTROY, INTRINSIC, etc.)
# Flag        string    EFFECT_FLAG_* noun (FIRE, COLD, LIGHT, HP, etc.)
# Flag2       string    EFFECT_FLAG_* noun word 2 (DOOR, TRAP, MONSTERS)
# Modifier    string    EFFECT_MOD_* adverb (LINE, BALL, AREA, RESIST, TIMED, etc.)
# Amount      XdY       damage/healing dice rolled per use
# Duration    float     for timed effects, how long (in game turns)
# Range       float     max range in tiles (0 = self only)
# Radius      float     AoE radius (0 = single target)

# --- HIT - damage / status attacks ---
